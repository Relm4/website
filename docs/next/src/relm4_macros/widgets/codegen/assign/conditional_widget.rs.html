1use quote::{quote, quote_spanned};
2use syn::Ident;
3
4use crate::widgets::{ConditionalBranches, ConditionalWidget, PropertyName};
5
6use super::AssignInfo;
7
8impl ConditionalWidget {
9    pub(super) fn assign_stream<'a>(
10        &'a self,
11        info: &mut AssignInfo<'a>,
12        p_name: &PropertyName,
13        sender_name: &'a Ident,
14    ) {
15        let assign_fn = p_name.assign_fn_stream(info);
16        let self_assign_args = p_name.assign_args_stream(info.widget_name);
17        let span = p_name.span();
18
19        let args = self.args.as_ref().map(|args| {
20            quote! {
21               , #args
22            }
23        });
24
25        let w_name = &self.name;
26        let assign_args = if let Some(assign_wrapper) = &self.assign_wrapper {
27            quote! { #assign_wrapper (&#w_name ) }
28        } else {
29            quote_spanned! { w_name.span() => &#w_name }
30        };
31
32        info.stream.extend(quote_spanned! {
33            span => #assign_fn(#self_assign_args #assign_args #args);
34        });
35
36        let mut info = AssignInfo {
37            stream: info.stream,
38            widget_name: &self.name,
39            template_path: None,
40            is_conditional: true,
41        };
42        match &self.branches {
43            ConditionalBranches::If(if_branches) => {
44                for branch in if_branches {
45                    let p_name = PropertyName::Ident(Ident::new("add_named", p_name.span()));
46                    branch.widget.assign_stream(&mut info, &p_name, sender_name);
47                }
48            }
49            ConditionalBranches::Match((_, _, match_arms)) => {
50                for arm in match_arms {
51                    let p_name = PropertyName::Ident(Ident::new("add_named", p_name.span()));
52                    arm.widget.assign_stream(&mut info, &p_name, sender_name);
53                }
54            }
55        }
56    }
57}
