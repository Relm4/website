23impl<C: AsyncFactoryComponent> AsyncFactoryBuilder<C>
24where
25    <C::ParentWidget as FactoryView>::ReturnedWidget: Clone,
26{
27    pub(super) fn new(init: C::Init, output_sender: Sender<C::Output>) -> Self {
28        // Used for all events to be processed by this component's internal service.
29        let (input_sender, input_receiver) = crate::channel::<C::Input>();
30
31        // Sends messages from commands executed from the background.
32        let (cmd_sender, cmd_receiver) = crate::channel::<C::CommandOutput>();
33
34        // Notifies the component's child commands that it is now deceased.
35        let (shutdown_notifier, shutdown_receiver) = shutdown::channel();
36
37        // Encapsulates the senders used by component methods.
38        let component_sender =
39            AsyncFactorySender::new(input_sender, output_sender, cmd_sender, shutdown_receiver);
40
41        let root_widget = C::init_root();
42
43        Self {
44            init,
45            root_widget,
46            component_sender,
47            input_receiver,
48            cmd_receiver,
49            shutdown_notifier,
50        }
51    }
52
53    /// Starts the component, passing ownership to a future attached to a [gtk::glib::MainContext].
54    pub(super) fn launch(
55        self,
56        index: &DynamicIndex,
57        returned_widget: <C::ParentWidget as FactoryView>::ReturnedWidget,
58    ) -> AsyncFactoryHandle<C> {
59        let Self {
60            root_widget,
61            component_sender,
62            input_receiver,
63            cmd_receiver,
64            shutdown_notifier,
65            init,
66        } = self;
67
68        // Gets notifications when a component's model and view is updated externally.
69        let (notifier, notifier_receiver) = crate::channel();
70
71        let input_sender = component_sender.input_sender().clone();
72
73        let loading_widgets = C::init_loading_widgets(root_widget.clone());
74
75        let future_receiver = {
76            let index = index.clone();
77            let (future_sender, future_receiver) = crate::channel();
78
79            let future_data = FutureData {
80                shutdown_notifier,
81                index: index.clone(),
82                component_sender: component_sender.clone(),
83                root: root_widget.clone(),
84                returned_widget: returned_widget.clone(),
85                input_receiver,
86                cmd_receiver,
87                notifier_receiver,
88            };
89
90            crate::spawn_local(async move {
91                let data = C::init_model(init, &index, component_sender).await;
92                drop(loading_widgets);
93                let data_guard = future_data.start_runtime(data);
94                let _ = future_sender.send(data_guard);
95            });
96            future_receiver
97        };
98
99        let data = AsyncData::new(future_receiver);
100
101        // Give back a type for controlling the component service.
102        AsyncFactoryHandle {
103            data,
104            root_widget,
105            returned_widget,
106            input: input_sender,
107            notifier,
108        }
109    }
110}

125struct FutureData<C: AsyncFactoryComponent> {
126    shutdown_notifier: ShutdownSender,
127    index: DynamicIndex,
128    component_sender: AsyncFactorySender<C>,
129    root: C::Root,
130    returned_widget: <C::ParentWidget as FactoryView>::ReturnedWidget,
131    input_receiver: Receiver<C::Input>,
132    cmd_receiver: Receiver<C::CommandOutput>,
133    notifier_receiver: Receiver<()>,
134}
135
136impl<C: AsyncFactoryComponent> FutureData<C> {
137    fn start_runtime(self, data: C) -> DataGuard<C, C::Widgets, C::Output> {
138        let Self {
139            shutdown_notifier,
140            index,
141            component_sender,
142            root,
143            returned_widget,
144            cmd_receiver,
145            input_receiver,
146            notifier_receiver,
147        } = self;
148
149        let mut data = Box::new(data);
150        let widgets =
151            Box::new(data.init_widgets(&index, root, &returned_widget, component_sender.clone()));
152
153        let output_sender = component_sender.output_sender().clone();
154
155        // Spawns the component's service. It will receive both `Self::Input` and
156        // `Self::CommandOutput` messages. It will spawn commands as requested by
157        // updates, and send `Self::Output` messages externally.
158        DataGuard::new(
159            data,
160            widgets,
161            shutdown_notifier,
162            output_sender,
163            |mut model, mut widgets| async move {
164                let mut notifier = GuardedReceiver::new(notifier_receiver);
165                let mut cmd = GuardedReceiver::new(cmd_receiver);
166                let mut input = GuardedReceiver::new(input_receiver);
167                loop {
168                    futures::select!(
169                        // Performs the model update, checking if the update requested a command.
170                        // Runs that command asynchronously in the background using tokio.
171                        message = input => {
172                            let span = info_span!(
173                                "update_with_view",
174                                input=?message,
175                                component=any::type_name::<C>(),
176                                id=model.id(),
177                            );
178                            let _enter = span.enter();
179
180                            model.update_with_view(&mut widgets, message, component_sender.clone()).await;
181                        }
182
183                        // Handles responses from a command.
184                        message = cmd => {
185                            let span = info_span!(
186                                "update_cmd_with_view",
187                                cmd_output=?message,
188                                component=any::type_name::<C>(),
189                                id=model.id(),
190                            );
191                            let _enter = span.enter();
192
193                            model.update_cmd_with_view(&mut widgets, message, component_sender.clone()).await;
194                        }
195
196                        // Triggered when the model and view have been updated externally.
197                        _ = notifier => {
198                            model.update_view(&mut widgets, component_sender.clone());
199                        }
200                    );
201                }
202            },
203            C::shutdown,
204        )
205    }
