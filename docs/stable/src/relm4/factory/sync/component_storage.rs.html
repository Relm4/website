1use crate::factory::{FactoryComponent, FactoryView};

5#[derive(Debug)]
6pub(super) enum ComponentStorage<C: FactoryComponent> {
7    Builder(FactoryBuilder<C>),
8    Final(FactoryHandle<C>),
9}
10
11impl<C: FactoryComponent> ComponentStorage<C> {
12    pub(super) const fn get(&self) -> &C {
13        match self {
14            Self::Builder(builder) => &builder.data,
15            Self::Final(handle) => handle.data.get(),
16        }
17    }
18
19    pub(super) fn get_mut(&mut self) -> &mut C {
20        match self {
21            Self::Builder(builder) => &mut builder.data,
22            Self::Final(handle) => handle.data.get_mut(),
23        }
24    }
25
26    pub(super) const fn widget(&self) -> &C::Root {
27        match self {
28            Self::Builder(builder) => &builder.root_widget,
29            Self::Final(handle) => &handle.root_widget,
30        }
31    }
32
33    pub(super) fn send(&self, msg: C::Input) {
34        match self {
35            Self::Builder(builder) => builder.component_sender.input(msg),
36            Self::Final(handle) => handle.input.send(msg).unwrap(),
37        }
38    }
39
40    pub(super) fn state_change_notify(&self) {
41        if let Self::Final(handle) = self {
42            handle.notifier.send(()).unwrap();
43        }
44    }
45
46    pub(super) fn extract(self) -> C {
47        match self {
48            Self::Builder(builder) => *builder.data,
49            Self::Final(handle) => handle.data.into_inner(),
50        }
51    }
52
53    pub(super) fn launch(
54        self,
55        index: &C::Index,
56        returned_widget: <C::ParentWidget as FactoryView>::ReturnedWidget,
57    ) -> Option<Self> {
58        if let Self::Builder(builder) = self {
59            Some(Self::Final(builder.launch(index, returned_widget)))
60        } else {
61            None
62        }
63    }
64
65    pub(super) const fn returned_widget(
66        &self,
67    ) -> Option<&<C::ParentWidget as FactoryView>::ReturnedWidget> {
68        if let Self::Final(handle) = self {
69            Some(&handle.returned_widget)
70        } else {
71            None
72        }
73    }
74}
