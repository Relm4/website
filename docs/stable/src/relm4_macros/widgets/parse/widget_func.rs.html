8impl WidgetFunc {
9    pub(super) fn parse_with_path(input: ParseStream<'_>, path: &Path) -> Result<Self, ParseError> {
10        match Self::parse_with_path_internal(input, path) {
11            Ok(func) => Ok(func),
12            Err(err) => Err(err.add_path(path)),
13        }
14    }
15
16    fn parse_with_path_internal(input: ParseStream<'_>, path: &Path) -> Result<Self, ParseError> {
17        if input.peek(Ident) {
18            return Err(ParseError::Generic(
19                syn::Error::new(
20                    path.span()
21                        .join(input.span())
22                        .unwrap_or_else(|| input.span()),
23                    "A path must not be followed by an identifier",
24                )
25                .into_compile_error(),
26            )
27            .add_path(path));
28        }
29
30        let args = if input.peek(token::Paren) {
31            let paren_input = parse_util::parens(input)?;
32            Some(paren_input.call(Punctuated::parse_terminated)?)
33        } else {
34            None
35        };
36
37        let method_chain = if input.peek(token::Dot) {
38            let _dot: token::Dot = input.parse()?;
39            Some(Punctuated::parse_separated_nonempty(input)?)
40        } else {
41            None
42        };
43
44        let ty = if input.peek(Token! [->]) {
45            let _token: Token! [->] = input.parse()?;
46            Some(input.parse()?)
47        } else {
48            None
49        };
50
51        Ok(WidgetFunc {
52            path: path.clone(),
53            args,
54            method_chain,
55            ty,
56        })
57    }
58}
59
60impl WidgetFunc {
61    pub(super) fn parse(input: ParseStream<'_>) -> Result<Self, ParseError> {
62        let path = &input.parse()?;
63        Self::parse_with_path(input, path)
64    }
65}
