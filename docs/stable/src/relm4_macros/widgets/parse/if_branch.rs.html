1use proc_macro2::Span as Span2;
2use syn::parse::ParseStream;
3use syn::punctuated::Punctuated;
4use syn::token::{And, Paren};
5use syn::{Expr, ExprCall, ExprLit, ExprPath, Ident, Lit, LitStr};
6
7use crate::args::Args;
8use crate::widgets::{IfBranch, ParseError, Widget, parse_util};
9
10impl IfBranch {
11    pub(super) fn parse(input: ParseStream<'_>, index: usize) -> Result<Self, ParseError> {
12        let cond = input.parse()?;
13
14        let braced = parse_util::braces(input)?;
15
16        let attributes = braced.parse().ok();
17        let args = args_from_index(index, input.span());
18        let mut widget = Widget::parse(&braced, attributes, Some(args))?;
19        widget.ref_token = Some(And {
20            spans: [Span2::mixed_site()],
21        });
22
23        Ok(Self { cond, widget })
24    }
25}
26
27pub(super) fn args_from_index(index: usize, span: Span2) -> Args<Expr> {
28    Args {
29        inner: vec![Expr::Call(ExprCall {
30            attrs: Vec::new(),
31            func: Box::new(Expr::Path(ExprPath {
32                attrs: Vec::new(),
33                qself: None,
34                path: Ident::new("Some", span).into(),
35            })),
36            paren_token: Paren(span),
37            args: Punctuated::from_iter(vec![Expr::Lit(ExprLit {
38                attrs: Vec::new(),
39                lit: Lit::Str(LitStr::new(&format!("{index}"), span)),
40            })]),
41        })],
42    }
43}
