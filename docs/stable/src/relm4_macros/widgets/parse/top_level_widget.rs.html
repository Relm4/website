1use syn::parse::ParseStream;
2
3use crate::util;
4use crate::widgets::{
5    Attr, Attrs, Properties, Property, PropertyName, PropertyType, TopLevelWidget, Widget,
6    WidgetAttr, WidgetFunc, WidgetTemplateAttr, parse_util,
7};
8
9impl TopLevelWidget {
10    pub(super) fn parse(input: ParseStream<'_>) -> Self {
11        let attributes: Option<Attrs> = input.parse().ok();
12
13        // Look for #[root] attribute and remove it from the list if it exists
14        let (attributes, root_attr) = if let Some(prev_attributes) = attributes {
15            let mut attributes = Attrs {
16                inner: Vec::with_capacity(prev_attributes.inner.len()),
17            };
18            let mut root_attr = None;
19            for attr in prev_attributes.inner {
20                match attr {
21                    Attr::Root(ident) => {
22                        // Save root attribute and don't push it to the new list
23                        root_attr = Some(ident);
24                    }
25                    _ => attributes.inner.push(attr),
26                }
27            }
28            (Some(attributes), root_attr)
29        } else {
30            (None, None)
31        };
32
33        let inner = match Widget::parse(input, attributes, None) {
34            Ok(inner) => inner,
35            Err(err) => Widget {
36                doc_attr: None,
37                attr: WidgetAttr::None,
38                template_attr: WidgetTemplateAttr::None,
39                mutable: None,
40                name: parse_util::string_to_snake_case("incorrect_top_level_widget"),
41                name_assigned_by_user: false,
42                func: WidgetFunc {
43                    path: util::strings_to_path(&["gtk", "Box"]),
44                    args: None,
45                    method_chain: None,
46                    ty: None,
47                },
48                args: None,
49                properties: Properties {
50                    properties: vec![Property {
51                        name: PropertyName::Ident(parse_util::string_to_snake_case(
52                            "invalid_property",
53                        )),
54                        ty: PropertyType::ParseError(err),
55                    }],
56                },
57                assign_wrapper: None,
58                ref_token: None,
59                deref_token: None,
60                returned_widget: None,
61            },
62        };
63
64        Self { root_attr, inner }
65    }
66}
