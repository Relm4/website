11        let guard = if input.peek(token::FatArrow) {
12            None
13        } else {
14            Some((input.parse()?, input.parse()?))
15        };
16

18
19        let braced;

22            &braced
23        } else {
24            input
25        };
26

29
30        let ref_span = input.span();

32        widget.ref_token = Some(And { spans: [ref_span] });
33
34        // Parse trailing commas
35        if input.peek(Token![,]) {
36            let _comma: Token![,] = input.parse()?;
37        }
38
39        Ok(Self {
40            pattern,
41            guard,
42            arrow,
43            widget,
44        })
45    }
