6impl SignalHandler {
7    pub(super) fn parse_with_args(
8        input: ParseStream<'_>,
9        args: Option<Args<Expr>>,
10    ) -> Result<Self> {
11        let inner = if args.is_some() || input.peek(Token![move]) || input.peek(Token![|]) {
12            ClosureSignalHandler::parse_with_args(input, args).map(SignalHandlerVariant::Closure)
13        } else {
14            input.parse().map(SignalHandlerVariant::Expr)
15        }?;
16
17        let handler_id = if input.peek(Token![@]) {
18            let _arrow: Token![@] = input.parse()?;
19            input.parse()?
20        } else {
21            None
22        };
23
24        Ok(Self { inner, handler_id })
25    }
26}
