Trait Binding

pub trait Binding: Clone + IsA<Object> {
    type Target;

    // Required methods
    fn get(&self) -> Self::Target;
    fn set(&self, value: Self::Target);

    // Provided methods
    fn property_name() -> &'static str { ... }
    fn guard(&self) -> BindingGuard<Self> { ... }
}

Get a new BindingGuard from the object.
